! ===== LISTAS DE ACCESO (ACLs) =====

! ACL para bloquear WiFi Invitados (VLAN 70) de acceder a red interna
! (Esta VLAN está en Sucursal B pero bloqueamos tráfico aquí también)
ip access-list extended BLOQUEAR-INVITADOS
 remark *** Bloquear VLAN 70 Invitados de red interna ***
 deny ip 10.0.32.128 0.0.0.127 10.0.0.0 0.0.255.255
 permit tcp 10.0.32.128 0.0.0.127 any eq 80
 permit tcp 10.0.32.128 0.0.0.127 any eq 443
 permit udp 10.0.32.128 0.0.0.127 any eq 53
 permit udp any eq 67 any eq 68
 deny ip 10.0.32.128 0.0.0.127 any
 permit ip any any
 exit

! ACL para proteger acceso a impresoras (solo desde VLANs autorizadas)
ip access-list extended ACCESO-IMPRESORAS
 remark *** Control de acceso a impresoras VLAN 60 ***
 permit tcp 10.0.0.0 0.0.1.255 10.0.5.176 0.0.0.15 eq 9100
 permit tcp 10.0.5.64 0.0.0.63 10.0.5.176 0.0.0.15 eq 9100
 permit tcp 10.0.5.128 0.0.0.31 10.0.5.176 0.0.0.15 eq 9100
 permit tcp 10.0.5.160 0.0.0.15 10.0.5.176 0.0.0.15 eq 9100
 permit icmp any any echo
 permit icmp any any echo-reply
 deny ip any 10.0.5.176 0.0.0.15
 permit ip any any
 exit

! Aplicar ACL en interfaz correspondiente
interface GigabitEthernet0/0.60
 ip access-group ACCESO-IMPRESORAS in
 exit
