! ===== CONFIGURAR SPANNING TREE =====
! Configurar este switch como ROOT BRIDGE para todas las VLANs
spanning-tree mode rapid-pvst
spanning-tree vlan 10,20,30,40,60,80,90,99 priority 4096
spanning-tree vlan 10,20,30,40,60,80,90,99 root primary
spanning-tree portfast default
spanning-tree portfast bpduguard default

! ===== ETHERCHANNEL HACIA ROUTER CENTRAL =====
! Agregar enlaces redundantes al Router Central
interface range GigabitEthernet0/1-2
 description *** EtherChannel a R-Central ***
 channel-group 1 mode active
 no shutdown
 exit

interface Port-channel1
 description *** EtherChannel a R-Central ***
 switchport trunk encapsulation dot1q
 switchport mode trunk
 switchport trunk native vlan 99
 switchport trunk allowed vlan 10,20,30,40,60,80,90,99
 no shutdown
 exit
